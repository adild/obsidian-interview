// Abstract base class defining common methods and the template method
public abstract class BaseClass {
    // Common methods shared by all subclasses
    public void commonMethod1() {
        System.out.println("Common Method 1: Shared logic");
    }

    public void commonMethod2() {
        System.out.println("Common Method 2: Shared logic");
    }

    // Abstract method for unique behavior (to be implemented by subclasses)
    public abstract void uniqueMethod();

    // Template method defining the algorithm structure
    public final void templateMethod() {
        commonMethod1();
        uniqueMethod(); // Calls subclass-specific implementation
        commonMethod2();
    }
}

// Subclass 1
public class Class1 extends BaseClass {
    @Override
    public void uniqueMethod() {
        System.out.println("Class1: Unique method implementation");
    }
}

// Subclass 2
public class Class2 extends BaseClass {
    @Override
    public void uniqueMethod() {
        System.out.println("Class2: Unique method implementation");
    }
}

// Subclass 3
public class Class3 extends BaseClass {
    @Override
    public void uniqueMethod() {
        System.out.println("Class3: Unique method implementation");
    }
}

// Subclass 4
public class Class4 extends BaseClass {
    @Override
    public void uniqueMethod() {
        System.out.println("Class4: Unique method implementation");
    }
}

// Main class to demonstrate usage
public class Main {
    public static void main(String[] args) {
        BaseClass class1 = new Class1();
        BaseClass class2 = new Class2();
        BaseClass class3 = new Class3();
        BaseClass class4 = new Class4();

        System.out.println("Executing Class1:");
        class1.templateMethod();

        System.out.println("\nExecuting Class2:");
        class2.templateMethod();

        System.out.println("\nExecuting Class3:");
        class3.templateMethod();

        System.out.println("\nExecuting Class4:");
        class4.templateMethod();
    }
}
